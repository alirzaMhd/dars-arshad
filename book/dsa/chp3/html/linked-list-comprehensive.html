Singly Linked List

Node Structure:
struct Node { Type data; Node* next; }
Each node contains data and a pointer to the next node

1. Search Operation

Node* Search(Node* head, Type key): current = head while current != NULL: if current.data == key: return current current = current.next return NULL // not found

2. Insert Operation

// Insert at beginning void InsertFront(Node* &head, Type data): newNode = new Node(data) newNode.next = head head = newNode // Insert after given node void InsertAfter(Node* prev, Type data): newNode = new Node(data) newNode.next = prev.next prev.next = newNode // Insert at end void InsertEnd(Node* &head, Type data): newNode = new Node(data) if head == NULL: head = newNode return current = head while current.next != NULL: current = current.next current.next = newNode

3. Delete Operation

// Delete from beginning void DeleteFront(Node* &head): if head == NULL: return temp = head head = head.next delete temp // Delete after given node void DeleteAfter(Node* prev): if prev == NULL or prev.next == NULL: return temp = prev.next prev.next = temp.next delete temp // Delete specific value void Delete(Node* &head, Type key): if head == NULL: return if head.data == key: DeleteFront(head) return prev = head while prev.next != NULL: if prev.next.data == key: DeleteAfter(prev) return prev = prev.next

Search

Time: O(n)

Space: O(1)

Insert (at head)

Time: O(1)

Space: O(1)

Delete (at head)

Time: O(1)

Space: O(1)

Doubly Linked List

Node Structure:
struct Node { Type data; Node* prev; Node* next; }
Each node has pointers to both previous and next nodes

1. Search Operation

Node* Search(Node* head, Type key): current = head while current != NULL: if current.data == key: return current current = current.next return NULL

2. Insert Operation

// Insert at beginning void InsertFront(Node* &head, Type data): newNode = new Node(data) newNode.next = head newNode.prev = NULL if head != NULL: head.prev = newNode head = newNode // Insert after given node void InsertAfter(Node* prev, Type data): newNode = new Node(data) newNode.next = prev.next newNode.prev = prev if prev.next != NULL: prev.next.prev = newNode prev.next = newNode

3. Delete Operation

// Delete given node void Delete(Node* &head, Node* del): if head == NULL or del == NULL: return if head == del: head = del.next if del.next != NULL: del.next.prev = del.prev if del.prev != NULL: del.prev.next = del.next delete del

Advantages

• Can traverse backwards

• Delete in O(1) if node known

• No need to track previous node

Disadvantages

• Extra pointer field

• More memory per node

• More complex operations

// Insert at end (circular) void InsertEnd(Node* &head, Type data): newNode = new Node(data) if head == NULL: head = newNode newNode.next = head else: current = head while current.next != head: current = current.next current.next = newNode newNode.next = head // Delete from beginning (circular) void DeleteFront(Node* &head): if head == NULL: return if head.next == head: delete head head = NULL else: last = head while last.next != head: last = last.next temp = head head = head.next last.next = head delete temp

Advantages

• No NULL checks needed

• Circular traversal

• Good for round-robin scheduling

Applications

• Operating systems (CPU scheduling)

• Music playlists

• Multiplayer games (turn-based)

Avail List (Memory Management)

Concept:
Avail list = free list of unused nodes
Instead of calling new/delete repeatedly, maintain a pool of free nodes
How It Works
// Initialize avail list from array void Initialize(Node* avail, int size): for i = 0 to size-2: avail[i].next = &avail[i+1] avail[size-1].next = NULL // Get a node from avail list Node* GetNode(Node* &avail): if avail == NULL: return NULL // no free nodes temp = avail avail = avail.next return temp // Return node to avail list void FreeNode(Node* &avail, Node* node): node.next = avail avail = node

Polynomial Addition

PolyNode* Add(PolyNode* p1, PolyNode* p2): result = NULL while p1 != NULL and p2 != NULL: if p1.exp > p2.exp: InsertEnd(result, p1.coeff, p1.exp) p1 = p1.next elif p1.exp < p2.exp: InsertEnd(result, p2.coeff, p2.exp) p2 = p2.next else: // same exponent sum = p1.coeff + p2.coeff if sum != 0: InsertEnd(result, sum, p1.exp) p1 = p1.next p2 = p2.next // Add remaining terms while p1 != NULL: InsertEnd(result, p1.coeff, p1.exp) p1 = p1.next while p2 != NULL: InsertEnd(result, p2.coeff, p2.exp) p2 = p2.next return result
